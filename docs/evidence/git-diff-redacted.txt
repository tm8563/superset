+  border-bottom: 1px solid ${({ theme }: { theme: SupersetTheme }) => theme.colorBorderSecondary};

+  flex-shrink: 0;

+  color: ${({ theme }: { theme: SupersetTheme }) => theme.colorTextSecondary};

+  padding: 2px 6px;
+  font-size: 11px;
+  border-radius: 4px;
+  border: 1px solid ${({ theme }: { theme: SupersetTheme }) => theme.colorBorder};
+  background-color: ${({ theme }: { theme: SupersetTheme }) => theme.colorBgContainer};

+  display: flex;
+  flex-direction: column;

+  padding: 24px;
+  color: ${({ theme }: { theme: SupersetTheme }) => theme.colorTextSecondary};

+  display: flex;
+  align-items: center;
+  justify-content: center;
+  height: 100%;
+  font-size: 13px;
+  color: ${({ theme }: { theme: SupersetTheme }) => theme.colorTextSecondary};
+`;
+
+const ErrorStateContainer = styled.div<{ theme?: SupersetTheme }>`
+  display: flex;
+  align-items: center;
+  justify-content: center;
+  height: 100%;
+  padding: 16px;
+  color: ${({ theme }: { theme: SupersetTheme }) => theme.colorError};
+  font-size: 13px;
+`;
+
+// ============================================================================
+// MAIN COMPONENT
+// ============================================================================
+
+export const EnterpriseTableChart: React.FC<EnterpriseTableTransformedProps> = ({
+  data,
+  columns,
+  width,
+  height,
+  isLoading = false,
+  errorMessage,
+  pageSize: initialPageSize = 20,
+  includeSearch = true,
+  enableColumnSort = true,
+  enableColumnResize = true,
+  enableColumnReorder = true,
+  enableColumnFilters = true,
+}) => {
+  // --- 1. State ---
+  const [searchTerm, setSearchTerm] = useState<string>('');
+  const [sortConfig, setSortConfig] = useState<ColumnSortItem[]>([]);
+  const [columnOrder, setColumnOrder] = useState<string[]>([]);
+  const [columnWidths, setColumnWidths] = useState<Record<string, number>>({});
+  const [columnFilters, setColumnFilters] = useState<Record<string, ColumnFilter>>({});
+  const [activeFilterPopover, setActiveFilterPopover] = useState<string | null>(null);
+  const [currentPage, setCurrentPage] = useState<number>(0);
+  const [pageSize, setPageSize] = useState<number>(initialPageSize);
+
+  // Initialize and synchronize column order when columns change
+  useEffect(() => {
+    const colKeys = columns.map(c => c.key);
+    setColumnOrder(prev => {
+      if (prev.length === 0 || !colKeys.every(k => prev.includes(k))) {
+        return colKeys;
+      }
+      return prev;
+    });
+  }, [columns]);
+
+  // Order columns according to columnOrder state
+  const orderedColumns = useMemo(() => {
+    if (columnOrder.length === 0) return columns;
+    const colMap = new Map(columns.map(c => [c.key, c]));
+    const result: typeof columns = [];
+    columnOrder.forEach(k => {
+      const col = colMap.get(k);
+      if (col) result.push(col);
+    });
+    // Add any newly added columns that might not be in columnOrder yet
+    columns.forEach(col => {
+      if (!result.some(c => c.key === col.key)) {
+        result.push(col);
+      }
+    });
+    return result;
+  }, [columns, columnOrder]);
+
+  // --- 2. Multi-Column Sorting Logic ---
+  const handleSort = useCallback(
+    (colKey: string, isShiftPressed: boolean) => {
+      if (!enableColumnSort) return;
+      setSortConfig(prevSorts => {
+        const existingIndex = prevSorts.findIndex(s => s.key === colKey);
+        if (isShiftPressed) {
+          // Multi-sort: cycle through asc -> desc -> remove
+          if (existingIndex === -1) {
+            return [...prevSorts, { key: colKey, direction: 'asc' }];
+          }
+          const existing = prevSorts[existingIndex];
+          if (existing.direction === 'asc') {
+            const next = [...prevSorts];
+            next[existingIndex] = { key: colKey, direction: 'desc' };
+            return next;
+          }
+          // Remove from multi-sort
+          return prevSorts.filter(s => s.key !== colKey);
+        }
+
+        // Single sort:
+        if (existingIndex !== -1 && prevSorts.length === 1) {
+          const existing = prevSorts[0];
+          if (existing.direction === 'asc') {
+            return [{ key: colKey, direction: 'desc' }];
+          }
+          return [];
+        }
+        return [{ key: colKey, direction: 'asc' }];
+      });
+      setCurrentPage(0);
+    },
+    [enableColumnSort],
+  );
+
+  // --- 3. Column Reorder Logic ---
+  const handleMoveColumn = useCallback(
+    (colKey: string, direction: 'left' | 'right') => {
+      if (!enableColumnReorder) return;
+      setColumnOrder(prevOrder => {
+        const currentIndex = prevOrder.indexOf(colKey);
+        if (currentIndex === -1) return prevOrder;
+        const targetIndex = direction === 'left' ? currentIndex - 1 : currentIndex + 1;
+        if (targetIndex < 0 || targetIndex >= prevOrder.length) return prevOrder;
+
+        const newOrder = [...prevOrder];
+        const temp = newOrder[currentIndex];
+        newOrder[currentIndex] = newOrder[targetIndex];
+        newOrder[targetIndex] = temp;
+        return newOrder;
+      });
+    },
+    [enableColumnReorder],
+  );
+
+  // --- 4. Column Resizing Logic ---
+  const handleResizeMouseDown = useCallback(
+    (colKey: string, e: ReactMouseEvent) => {
+      if (!enableColumnResize) return;
+      e.preventDefault();
+      e.stopPropagation();
+
+      const startX = e.clientX;
+      const startWidth = columnWidths[colKey] || 120;
+
+      const handleMouseMove = (moveEvent: MouseEvent) => {
+        const delta = moveEvent.clientX - startX;
+        const newWidth = Math.max(60, Math.min(800, startWidth + delta));
+        setColumnWidths(prev => ({
+          ...prev,
+          [colKey]: newWidth,
+        }));
+      };
+
+      const handleMouseUp = () => {
+        window.removeEventListener('mousemove', handleMouseMove);
+        window.removeEventListener('mouseup', handleMouseUp);
+      };
+
+      window.addEventListener('mousemove', handleMouseMove);
+      window.addEventListener('mouseup', handleMouseUp);
+    },
+    [columnWidths, enableColumnResize],
+  );
+
+  // --- 5. Column Filtering Logic ---
+  const handleSetColumnFilter = useCallback(
+    (colKey: string, operator: FilterOperator, value: string, secondaryValue?: string) => {
+      setColumnFilters(prev => ({
+        ...prev,
+        [colKey]: { key: colKey, operator, value, secondaryValue },
+      }));
+      setCurrentPage(0);
+    },
+    [],
+  );
+
+  const handleClearColumnFilter = useCallback((colKey: string) => {
+    setColumnFilters(prev => {
+      const next = { ...prev };
+      delete next[colKey];
+      return next;
+    });
+    setCurrentPage(0);
+  }, []);
+
+  const handleResetAll = useCallback(() => {
+    setSearchTerm('');
+    setSortConfig([]);
+    setColumnFilters({});
+    setColumnOrder(columns.map(c => c.key));
+    setCurrentPage(0);
+  }, [columns]);
+
+  // --- 6. Data Filtering & Sorting Pipeline ---
+  const processedData = useMemo(() => {
+    let result = [...data];
+
+    // A. Global Substring Search Filter
+    if (searchTerm.trim()) {
+      const lower = searchTerm.toLowerCase();
+      result = result.filter(row =>
+        Object.values(row).some(
+          val => val !== null && val !== undefined && String(val).toLowerCase().includes(lower),
+        ),
+      );
+    }
+
+    // B. Per-Column Filters
+    Object.values(columnFilters).forEach(filter => {
+      if (!filter.value && filter.operator !== 'between') return;
+
+      result = result.filter(row => {
+        const rawVal = row[filter.key];
+        if (rawVal === null || rawVal === undefined) return false;
+
+        const valStr = String(rawVal).toLowerCase();
+        const filterValStr = String(filter.value).toLowerCase();
+
+        switch (filter.operator) {
+          case 'contains':
+            return valStr.includes(filterValStr);
+          case 'equals':
+            if (typeof rawVal === 'number') {
+              return Number(rawVal) === Number(filter.value);
+            }
+            return valStr === filterValStr;
+          case 'startsWith':
+            return valStr.startsWith(filterValStr);
+          case 'greaterThan':
+            return Number(rawVal) > Number(filter.value);
+          case 'lessThan':
+            return Number(rawVal) < Number(filter.value);
+          case 'between': {
+            const num = Number(rawVal);
+            const min = Number(filter.value);
+            const max = Number(filter.secondaryValue ?? filter.value);
+            return num >= min && num <= max;
+          }
+          default:
+            return true;
+        }
+      });
+    });
+
+    // C. Multi-Column Sorting
+    if (sortConfig.length > 0) {
+      result.sort((a, b) => {
+        for (const sort of sortConfig) {
+          const valA = a[sort.key];
+          const valB = b[sort.key];
+
+          if (valA === valB) continue;
+          if (valA === null || valA === undefined) return 1;
+          if (valB === null || valB === undefined) return -1;
+
+          let comp = 0;
+          if (typeof valA === 'number' && typeof valB === 'number') {
+            comp = valA - valB;
+          } else {
+            comp = String(valA).localeCompare(String(valB));
+          }
+
+          if (comp !== 0) {
+            return sort.direction === 'asc' ? comp : -comp;
+          }
+        }
+        return 0;
+      });
+    }
+
+    return result;
+  }, [data, searchTerm, columnFilters, sortConfig]);
+
+  // --- 7. Pagination Slicing ---
+  const totalEntries = processedData.length;
+  const totalPages = Math.ceil(totalEntries / pageSize) || 1;
+  const safeCurrentPage = Math.min(currentPage, totalPages - 1);
+
+  const paginatedData = useMemo(() => {
+    const start = safeCurrentPage * pageSize;
+    return processedData.slice(start, start + pageSize);
+  }, [processedData, safeCurrentPage, pageSize]);
+
+  const hasActiveCustomizations =
+    searchTerm !== '' ||
+    sortConfig.length > 0 ||
+    Object.keys(columnFilters).length > 0;
+
+  // --- 8. Render States ---
+  if (isLoading) {
+    return (
+      <Container width={width} height={height} data-test="enterprise-table-loading">
+        <LoadingStateContainer>{t('Loading Enterprise Table...')}</LoadingStateContainer>
+      </Container>
+    );
+  }
+
+  if (errorMessage) {
+    return (
+      <Container width={width} height={height} data-test="enterprise-table-error">
+        <ErrorStateContainer>{errorMessage}</ErrorStateContainer>
+      </Container>
+    );
+  }
+
+  if (!data || data.length === 0 || columns.length === 0) {
+    return (
+      <Container width={width} height={height} data-test="enterprise-table-empty">
+        <EmptyStateContainer>{t('No data records available')}</EmptyStateContainer>
+      </Container>
+    );
+  }
+
+  return (
+    <Container width={width} height={height} data-test="enterprise-table-container">
+      {/* Top Header Controls Bar */}
+      <HeaderBar>
+        <TitleContainer>
+          <Title>{t('Enterprise Interactive Table')}</Title>
+          <Badge>Phase 3 MVP</Badge>
+          <Badge>{t('%s rows', totalEntries)}</Badge>
+        </TitleContainer>
+
+        <ControlsGroup>
+          {includeSearch && (
+            <SearchInput
+              type="text"
+              placeholder={t('Search table...')}
+              value={searchTerm}
+              onChange={(e: ChangeEvent<HTMLInputElement>) => {
+                setSearchTerm(e.target.value);
+                setCurrentPage(0);
+              }}
+              data-test="enterprise-table-search"
+            />
+          )}
+          {hasActiveCustomizations && (
+            <ActionButton
+              onClick={handleResetAll}
+              data-test="reset-filters-btn"
+              title={t('Reset search, sort, and column filters')}
+            >
+              {t('Reset Filters')}
+            </ActionButton>
+          )}
+        </ControlsGroup>
+      </HeaderBar>
+
+      {/* Main Grid View */}
+      <TableWrapper>
+        <StyledTable data-test="enterprise-table-grid">
+          <TableHead>
+            <tr>
+              {orderedColumns.map((col, colIdx) => {
+                const sortItem = sortConfig.find(s => s.key === col.key);
+                const sortIndex = sortConfig.findIndex(s => s.key === col.key);
+                const isFiltered = Boolean(columnFilters[col.key]);
+                const isPopoverOpen = activeFilterPopover === col.key;
+                const colWidth = columnWidths[col.key];
+
+                return (
+                  <HeaderCell
+                    key={col.key}
+                    width={colWidth}
+                    isMetric={col.isMetric}
+                    data-test={`header-${col.key}`}
+                  >
+                    <HeaderContent isMetric={col.isMetric}>
+                      {/* Left/Right Reorder Buttons */}
+                      {enableColumnReorder && (
+                        <HeaderIcons>
+                          <IconButton
+                            disabled={colIdx === 0}
+                            onClick={() => handleMoveColumn(col.key, 'left')}
+                            data-test={`reorder-left-${col.key}`}
+                            title={t('Move column left')}
+                          >
+                            ◀
+                          </IconButton>
+                          <IconButton
+                            disabled={colIdx === orderedColumns.length - 1}
+                            onClick={() => handleMoveColumn(col.key, 'right')}
+                            data-test={`reorder-right-${col.key}`}
+                            title={t('Move column right')}
+                          >
+                            ▶
+                          </IconButton>
+                        </HeaderIcons>
+                      )}
+
+                      {/* Header Label and Sort Toggle */}
+                      <HeaderLabel
+                        onClick={(e: ReactMouseEvent) => handleSort(col.key, e.shiftKey)}
+                        data-test={`sort-btn-${col.key}`}
+                        title={
+                          enableColumnSort
+                            ? t('Click to sort, Shift+Click for multi-column sort')
+                            : undefined
+                        }
+                      >
+                        {col.label}
+                      </HeaderLabel>
+
+                      {/* Sort Direction Indicator */}
+                      {sortItem && (
+                        <SortIndicator data-test={`sort-indicator-${col.key}`}>
+                          {sortItem.direction === 'asc' ? '▲' : '▼'}
+                          {sortConfig.length > 1 && (
+                            <SortPriority>{sortIndex + 1}</SortPriority>
+                          )}
+                        </SortIndicator>
+                      )}
+
+                      {/* Per-Column Filter Button */}
+                      {enableColumnFilters && (
+                        <IconButton
+                          active={isFiltered || isPopoverOpen}
+                          onClick={() =>
+                            setActiveFilterPopover(isPopoverOpen ? null : col.key)
+                          }
+                          data-test={`filter-toggle-${col.key}`}
+                          title={t('Filter column')}
+                        >
+                          {isFiltered ? '●' : '🔍'}
+                        </IconButton>
+                      )}
+                    </HeaderContent>
+
+                    {/* In-Header Filter Popover */}
+                    {isPopoverOpen && (
+                      <FilterPopover
+                        isMetric={col.isMetric}
+                        data-test={`filter-popover-${col.key}`}
+                      >
+                        <FilterSelect
+                          value={
+                            columnFilters[col.key]?.operator ??
+                            (col.isMetric ? 'greaterThan' : 'contains')
+                          }
+                          onChange={(e: ChangeEvent<HTMLSelectElement>) => {
+                            const op = e.target.value as FilterOperator;
+                            const currentVal = columnFilters[col.key]?.value ?? '';
+                            handleSetColumnFilter(col.key, op, currentVal);
+                          }}
+                          data-test={`filter-operator-${col.key}`}
+                        >
+                          {col.isMetric ? (
+                            <>
+                              <option value="greaterThan">{t('Greater than (> )')}</option>
+                              <option value="lessThan">{t('Less than (<)')}</option>
+                              <option value="equals">{t('Equals (=)')}</option>
+                              <option value="between">{t('Between')}</option>
+                            </>
+                          ) : (
+                            <>
+                              <option value="contains">{t('Contains')}</option>
+                              <option value="equals">{t('Equals')}</option>
+                              <option value="startsWith">{t('Starts with')}</option>
+                            </>
+                          )}
+                        </FilterSelect>
+
+                        <FilterInput
+                          type={col.isMetric ? 'number' : 'text'}
+                          placeholder={t('Filter value...')}
+                          value={columnFilters[col.key]?.value ?? ''}
+                          onChange={(e: ChangeEvent<HTMLInputElement>) => {
+                            const currentOp =
+                              columnFilters[col.key]?.operator ??
+                              (col.isMetric ? 'greaterThan' : 'contains');
+                            handleSetColumnFilter(
+                              col.key,
+                              currentOp,
+                              e.target.value,
+                              columnFilters[col.key]?.secondaryValue,
+                            );
+                          }}
+                          data-test={`filter-input-${col.key}`}
+                        />
+
+                        {columnFilters[col.key]?.operator === 'between' && (
+                          <FilterInput
+                            type="number"
+                            placeholder={t('Max value...')}
+                            value={columnFilters[col.key]?.secondaryValue ?? ''}
+                            onChange={(e: ChangeEvent<HTMLInputElement>) => {
+                              handleSetColumnFilter(
+                                col.key,
+                                'between',
+                                columnFilters[col.key]?.value ?? '',
+                                e.target.value,
+                              );
+                            }}
+                            data-test={`filter-secondary-input-${col.key}`}
+                          />
+                        )}
+
+                        <FilterActions>
+                          <ActionButton
+                            onClick={() => {
+                              handleClearColumnFilter(col.key);
+                              setActiveFilterPopover(null);
+                            }}
+                            data-test={`filter-clear-${col.key}`}
+                          >
+                            {t('Clear')}
+                          </ActionButton>
+                          <ActionButton
+                            active
+                            onClick={() => setActiveFilterPopover(null)}
+                            data-test={`filter-apply-${col.key}`}
+                          >
+                            {t('Close')}
+                          </ActionButton>
+                        </FilterActions>
+                      </FilterPopover>
+                    )}
+
+                    {/* Column Resizing Handle */}
+                    {enableColumnResize && (
+                      <ResizeHandle
+                        onMouseDown={(e: ReactMouseEvent) =>
+                          handleResizeMouseDown(col.key, e)
+                        }
+                        data-test={`resize-handle-${col.key}`}
+                      />
+                    )}
+                  </HeaderCell>
+                );
+              })}
+            </tr>
+          </TableHead>
+          <tbody>
+            {paginatedData.map((row: DataRecord, rowIndex: number) => (
+              <TableRow key={`row-${rowIndex}`} data-test={`row-${rowIndex}`}>
+                {orderedColumns.map(col => {
+                  const val = row[col.key];
+                  const colWidth = columnWidths[col.key];
+                  const displayVal =
+                    val === null || val === undefined ? '-' : String(val);
+                  return (
+                    <Cell
+                      key={`cell-${col.key}-${rowIndex}`}
+                      width={colWidth}
+                      isMetric={col.isMetric}
+                      data-test={`cell-${col.key}-${rowIndex}`}
+                    >
+                      {displayVal}
+                    </Cell>
+                  );
+                })}
+              </TableRow>
+            ))}
+          </tbody>
+        </StyledTable>
+      </TableWrapper>
+
+      {/* Bottom Pagination Footer */}
+      <PaginationFooter data-test="pagination-footer">
+        <PaginationInfo>
+          {t(
+            'Showing %s to %s of %s entries',
+            totalEntries === 0 ? 0 : safeCurrentPage * pageSize + 1,
+            Math.min((safeCurrentPage + 1) * pageSize, totalEntries),
+            totalEntries,
+          )}
+          {totalEntries !== data.length && (
+            <span> {t('(filtered from %s total entries)', data.length)}</span>
+          )}
+        </PaginationInfo>
+
+        <PaginationControls>
+          <PageSizeSelect
+            value={pageSize}
+            onChange={(e: ChangeEvent<HTMLSelectElement>) => {
+              setPageSize(Number(e.target.value));
+              setCurrentPage(0);
+            }}
+            data-test="page-size-select"
+          >
+            <option value={10}>10 / page</option>
+            <option value={20}>20 / page</option>
+            <option value={50}>50 / page</option>
+            <option value={100}>100 / page</option>
+            <option value={200}>200 / page</option>
+          </PageSizeSelect>
+
+          <ActionButton
+            disabled={safeCurrentPage === 0}
+            onClick={() => setCurrentPage(0)}
+            data-test="page-first"
+            title={t('First page')}
+          >
+            ⏮
+          </ActionButton>
+          <ActionButton
+            disabled={safeCurrentPage === 0}
+            onClick={() => setCurrentPage(prev => Math.max(0, prev - 1))}
+            data-test="page-prev"
+            title={t('Previous page')}
+          >
+            ◀
+          </ActionButton>
+
+          <span data-test="page-indicator">
+            {t('Page %s of %s', safeCurrentPage + 1, totalPages)}
+          </span>
+
+          <ActionButton
+            disabled={safeCurrentPage >= totalPages - 1}
+            onClick={() => setCurrentPage(prev => Math.min(totalPages - 1, prev + 1))}
+            data-test="page-next"
+            title={t('Next page')}
+          >
+            ▶
+          </ActionButton>
+          <ActionButton
+            disabled={safeCurrentPage >= totalPages - 1}
+            onClick={() => setCurrentPage(totalPages - 1)}
+            data-test="page-last"
+            title={t('Last page')}
+          >
+            ⏭
+          </ActionButton>
+        </PaginationControls>
+      </PaginationFooter>
+    </Container>
+  );
+};
+
+export default EnterpriseTableChart;

+  EnterpriseTableColumn,

+  EnterpriseTableTransformedProps,
+} from './types';
+
+export function transformProps(chartProps: ChartProps): EnterpriseTableTransformedProps {
+  const { width, height, formData, rawFormData, queriesData } = chartProps;
+  const typedFormData = (rawFormData ?? formData) as EnterpriseTableFormData;
+
+  const data: DataRecord[] = (queriesData[0]?.data as DataRecord[]) ?? [];
+  const colnames: string[] = queriesData[0]?.colnames ?? [];
+
+  const metricLabels = new Set(
+    (typedFormData.metrics ?? []).map(m => getMetricLabel(m)),
+  );
+
+  const columns: EnterpriseTableColumn[] = colnames.map(colName => {
+    const isMetric = metricLabels.has(colName);
+    const sampleVal = data.find(
+      row => row[colName] !== null && row[colName] !== undefined,
+    )?.[colName];
+    const isNumeric = isMetric || typeof sampleVal === 'number';
+
+    return {
+      key: colName,
+      label: getColumnLabel(colName),
+      isMetric,
+      dataType: isNumeric ? 'number' : 'string',
+    };
+  });
+
+  return {
+    data,
+    columns,
+    width,
+    height,
+    pageSize: typedFormData.page_size ?? 20,
+    includeSearch: Boolean(typedFormData.include_search),
+    enableColumnSort: typedFormData.enable_column_sort !== false,
+    enableColumnResize: typedFormData.enable_column_resize !== false,
+    enableColumnReorder: typedFormData.enable_column_reorder !== false,
+    enableColumnFilters: typedFormData.enable_column_filters !== false,
+    rawFormData: typedFormData,
+  };
